cmake_minimum_required(VERSION 3.20)
project(eventbook LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(eventbook_core cpp/src/order_book.cpp)
set_target_properties(eventbook_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(eventbook_core PUBLIC cpp/include)
enable_testing()
add_executable(order_book_test cpp/tests/order_book_test.cpp)
target_link_libraries(order_book_test PRIVATE eventbook_core)
add_test(NAME order_book_test COMMAND order_book_test)

option(EVENTBOOK_BUILD_PYTHON_BINDINGS "Build the pybind11 order-book module" OFF)
if(EVENTBOOK_BUILD_PYTHON_BINDINGS)
  set(PYBIND11_FINDPYTHON ON)
  find_package(Python 3.12 COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_eventbook_core cpp/src/python_bindings.cpp)
  target_link_libraries(_eventbook_core PRIVATE eventbook_core)
  target_include_directories(_eventbook_core PRIVATE cpp/include)
  add_test(NAME python_bindings_test COMMAND ${Python_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/cpp/tests/python_bindings_test.py)
  set_tests_properties(python_bindings_test PROPERTIES ENVIRONMENT "PYTHONPATH=$<TARGET_FILE_DIR:_eventbook_core>")
endif()
